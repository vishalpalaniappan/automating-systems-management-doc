\subsubsection*{Meaning of Add}

\begin{minted}[linenos, frame=lines, mathescape, breaklines=true]{python}
    design("add")
    type("primitive")

    description("Adds two numbers and produces a result. The numbers must have type float or int and the result will be float or int type.")

    narrative_structure {
        # Example: store result of add cost_of_bagel and cost_of_coffee in total_cost
        {
            syntax: "store result of add {addend1} and {addend2} in {resultName}"
        }
    }

    participants {
        participant(name="addend1", required="true")
        participant(name="addend2", required="true")
        participant(name="resultName", required="true")
    }

    returns {
        participant(
            name="result",
            type="float" or "int",
            description="Result of the addition"
        )
    }

    behavior add {
        # Get the addends from the world state
        getFromWorldState(name="addend1", role="addend1")
        getFromWorldState(name="addend2", role="addend2")
        getFromWorldState(name="resultName", role="resultName")

        # adds the two values and store result in the value of resultName 
        _adds(result, addend1, addend2)

        # Update the world state
        updateWorldState(
            name=resultName,
            value=result,
            type="float" or "int"
        )
    }

    run (add)


    invariants {
        assert({
            id:1,
            participants: ["addend1"],
            property: ["type"]
            condition: "float" || "int"
            description: "Addend1 must be a whole number or float.",
            invalidValues: [
                {
                    value: "string_value"
                    description: "A string is provided as an addend."
                }
            ]
        }),
        assert({
            id:2,
            participants: ["addend2"],
            property: ["type"]
            condition: "float" || "int"
            description: "Addend2 must be a whole number or float.",
            invalidValues: [
                {
                    value: "string_value"
                    description: "A string is provided as an addend."
                }
            ]
        })
    }

    generate{
        {
            type: "success",
            sentence: f"Added {addend1} to {addend2} and resulted in {result}.",
        },
        {
            type: "fail_id_1",
            sentence: f"Failed addition because addend1 was not a float or int.",
        },
        {
            type: "fail_id_2",
            sentence: f"Failed addition because addend2 was not a float or int.",
        }
    }
\end{minted}

\begin{minted}[linenos, frame=lines, mathescape, breaklines=true]{python}
def add(addend1, addend2):
    return addend1 + addend2
\end{minted}
